public dialogService = inject(MozDialogService)

dialogResult = '';

openDataDialog() {
const dialogRef = this.dialogService.open(DialogDataContent, {
width: '500px',
data: { message:'Hello from Doc!' },
disableClose: true
});

dialogRef.afterClosed().subscribe((result: any) => {
if (result) {
this.dialogResult = result;
}
});
};
